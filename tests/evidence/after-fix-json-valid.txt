# команда: ./mbparse "<hex>" -j | tests/jsoncheck

==================== 1. вывод CLI (`-j`) ====================

$ ./mbparse "01 03 00 00 00 0A C5 CD" -j

$ ./mbparse "01 83 02 C0 F1" -j
{"protocol":"RTU","slave_id":1,"crc":61888,"crc_ok":true,"pdu":{"slave_id":1,"function":131,"function_name":"Unknown Function Code","is_exception":true,"exception_code":2,"exception_name":"Illegal Data Address","data_len":0,"data":[]}}

$ ./mbparse "00 01 00 00 00 06 01 03 00 00 00 0A" -j
{"protocol":"TCP","transaction_id":1,"protocol_id":0,"length":6,"unit_id":1,"pdu":{"slave_id":1,"function":3,"function_name":"Read Holding Registers","is_exception":false,"data_len":4,"data":[0,0,0,10]}}

$ printf '%s\n' "01 03 00 00 00 0A C5 CD" | ./mbparse - -j          # stdin
